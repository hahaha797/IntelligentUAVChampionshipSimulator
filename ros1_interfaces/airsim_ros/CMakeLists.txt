cmake_minimum_required(VERSION 3.0.2)
project(airsim_ros)
find_package(catkin REQUIRED COMPONENTS message_generation std_msgs geometry_msgs geographic_msgs)
file(GLOB MSG_FILES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/msg" "msg/*.msg")
file(GLOB SRV_FILES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/srv" "srv/*.srv")
add_message_files(FILES ${MSG_FILES})
add_service_files(FILES ${SRV_FILES})
generate_messages(DEPENDENCIES std_msgs geometry_msgs geographic_msgs)
catkin_package(CATKIN_DEPENDS message_runtime std_msgs geometry_msgs geographic_msgs)
